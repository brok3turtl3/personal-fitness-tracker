    ```typescript
    import { Component, EventEmitter, Input, Output } from '@angular/core';
    import { CommonModule } from '@angular/common';
    import { ErrorStateComponent } from './error-state.component';

    /**
     * Migration-failure recovery banner (D-15).
     *
     * Thin wrapper composing <app-error-state> with three actions:
     *   - Retry migration
     *   - Copy backup JSON to clipboard
     *   - Continue with empty data
     *
     * Inputs surface the recovery-key payload (key, fromVersion, toVersion)
     * so the user can locate their backup. AppComponent wires the three
     * outputs to StorageService methods.
     *
     * Per RESEARCH §"Open Q 3": separate component (not inline in AppComponent)
     * to isolate clipboard fallback logic and keep AppComponent simple.
     */
    @Component({
      selector: 'app-recovery-banner',
      standalone: true,
      imports: [CommonModule, ErrorStateComponent],
      template: `
        <app-error-state
          title="Couldn't upgrade your stored data"
          [message]="recoveryMessage()"
        >
          <button type="button" class="btn btn-primary" (click)="retry.emit()">
            Retry migration
          </button>
          <button type="button" class="btn btn-secondary" (click)="onCopyBackup()">
            Copy backup JSON to clipboard
          </button>
          <button type="button" class="btn btn-tertiary" (click)="continueEmpty.emit()">
            Continue with empty data
          </button>
        </app-error-state>

        @if (showFallbackTextarea) {
          <section class="recovery-banner__fallback">
            <p>Clipboard unavailable. Copy the JSON below manually:</p>
            <textarea readonly rows="10" class="recovery-banner__textarea">{{ backupJson }}</textarea>
          </section>
        }
      `,
      styles: [`
        .recovery-banner__fallback { margin-top: 1rem; }
        .recovery-banner__textarea { width: 100%; font-family: monospace; font-size: 0.85em; }
      `]
    })
    export class RecoveryBannerComponent {
      @Input({ required: true }) recoveryKey!: string;
      @Input({ required: true }) fromVersion!: number;
      @Input({ required: true }) toVersion!: number;
      /** Backup JSON text (read by AppComponent from LocalStorage and passed in). */
      @Input() backupJson = '';

      @Output() retry = new EventEmitter<void>();
      @Output() copyBackup = new EventEmitter<void>();
      @Output() continueEmpty = new EventEmitter<void>();

      showFallbackTextarea = false;

      recoveryMessage(): string {
        return `A backup of your data was saved at LocalStorage key '${this.recoveryKey}' before the upgrade was attempted (v${this.fromVersion} → v${this.toVersion}). You can retry, copy the backup for safekeeping, or continue with an empty dataset.`;
      }

      onCopyBackup(): void {
        // Feature-detect clipboard API (similar shape to id.ts Pitfall 5).
        if (typeof navigator !== 'undefined' && navigator.clipboard?.writeText) {
          navigator.clipboard.writeText(this.backupJson).then(
            () => { this.copyBackup.emit(); },
            () => { this.showFallbackTextarea = true; this.copyBackup.emit(); },
          );
        } else {
          // Clipboard unavailable — show the textarea for manual copy.
          this.showFallbackTextarea = true;
          this.copyBackup.emit();
        }
      }
    }
    ```

      ```typescript
      private destroyRef = inject(DestroyRef);
      private storageService = inject(StorageService);

      migrationError: StorageError | null = null;
      recoveryKey = '';
      fromVersion = 0;
      toVersion = 0;
      backupJson = '';
      continuedEmpty = false;

      ngOnInit(): void {
        this.runInitialize();
      }

      private runInitialize(): void {
        this.storageService.initialize()
          .pipe(takeUntilDestroyed(this.destroyRef))
          .subscribe({
            next: () => { this.migrationError = null; },
            error: (e) => {
              if (e instanceof StorageError && e.code === 'MIGRATION_FAILED') {
                this.migrationError = e;
                this.parseRecoveryFromError(e);
                this.loadBackupJson();
              } else {
                // Non-migration storage errors are handled by individual feature pages
                // via <app-error-state>. Bubble through.
                console.error('AppComponent: storage init failed', e);
              }
            },
          });
      }

      private parseRecoveryFromError(e: StorageError): void {
        // Error message shape (from plan 09):
        //   "Migration failed (v3 → v4). Backup at fitness_tracker_data.backup.v3.<ts>."
        const match = /v(\d+)\s*→\s*v(\d+)\)\.\s*Backup at\s+(\S+)\.?$/.exec(e.message);
        if (match) {
          this.fromVersion = Number(match[1]);
          this.toVersion = Number(match[2]);
          this.recoveryKey = match[3];
        }
      }

      private loadBackupJson(): void {
        try {
          this.backupJson = localStorage.getItem(this.recoveryKey) ?? '';
        } catch {
          this.backupJson = '';
        }
      }

      onRetryMigration(): void {
        // Re-run initialize. Reset cached state so the storage service re-attempts.
        // Per CLAUDE.md: only StorageService touches localStorage. Use a public reset
        // method on StorageService if needed (else, set this.migrationError = null
        // and re-trigger; storage already throws on second initialize call without
        // reset since `initialized` flag stays false on the throw path per plan 09).
        this.runInitialize();
      }

      onCopyBackup(): void {
        // No-op: RecoveryBannerComponent handles the clipboard write internally.
      }

      onContinueEmpty(): void {
        this.continuedEmpty = true;
        this.migrationError = null;
        // The user accepts the empty-state path. AppComponent renders <router-outlet>;
        // feature pages will show empty states on first render.
      }
      ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-01 | Repudiation | User loses data without warning when migration fails | mitigate | Recovery banner is rendered ahead of `<router-outlet>`; the user MUST take an explicit action (Retry / Copy / Continue) before the app proceeds. Spec asserts banner blocks router-outlet rendering. |
| T-10-02 | Denial of Service | Clipboard API unavailable → user can't copy backup | mitigate | Feature detection: when `navigator.clipboard?.writeText` is unavailable, render a `<textarea>` with the backup JSON for manual copy. Spec covers the fallback path explicitly. |
| T-10-03 | Information Disclosure | Recovery banner exposes backup JSON in DOM | accept | Per RESEARCH §"Security Domain" — single-user local app; the user's own data on the user's own machine. The fallback `<textarea>` is `readonly`; user must intentionally copy. Aligns with collapsed `<details>` precedent in `<app-error-state>`. |
| T-10-04 | Tampering | "Continue with empty data" silently overwrites the user's data | accept | The migration write to LocalStorage is gated INSIDE storage.service.ts AFTER successful migrate (plan 09). On failure, the original data stays untouched (the new schema persistToStorage call never ran). "Continue with empty data" sets `this.continuedEmpty = true` in AppComponent → router-outlet renders → feature pages render empty states from cache only — they don't WRITE anything until the user interacts. The original LocalStorage `STORAGE_KEY` value plus the backup key are still intact for retry/copy. Acceptable because the user explicitly chose "Continue empty". |
