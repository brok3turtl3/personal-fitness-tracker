<!-- Identity-preserving update pattern (RESEARCH §Pattern 3) -->
updateX(id, input): Observable<X> {
  const v = validateX(input); if (!v.valid) return throwError(() => new XValidationError(v.errors));
  return this.storageService.getData().pipe(switchMap(data => {
    if (!data) return throwError(() => new Error('Storage not initialized'));
    const idx = data.arr.findIndex(s => s.id === id);
    if (idx < 0) return throwError(() => new Error('X not found'));
    const existing = data.arr[idx];
    const updated = { ...existing, ...input, id: existing.id, createdAt: existing.createdAt, updatedAt: new Date().toISOString() };
    const arr = [...data.arr]; arr[idx] = updated;
    return this.storageService.saveData({ ...data, arr }).pipe(map(() => updated));
  }));
}
