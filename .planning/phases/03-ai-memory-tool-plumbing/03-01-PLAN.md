4. `LegacyAppDataV4` + `LegacyChatConversationV4` + `LegacyChatMessageV4` typed shapes added to `legacy-schemas.ts` (FOUND-07 typed-legacy harness extension).
5. `migrateV4ToV5()` step appended to `storage.service.ts` migrate chain. Order: build `blocks` array → swap into new ChatMessage → the V5 shape simply has no `content` field. Defensive null guard: `text: msg.content ?? ''`.
6. Fixture pair `v4.json` / `v5-expected.json` + 4-case malformed-V4 matrix.
7. Storage spec extension with 5+ scenarios (V4→V5 happy path, V4→V5 idempotent on V5 input, malformed throws MIGRATION_FAILED with recovery key, content lift preserves text, content field removed from V5).
8. **Compile-saving cut-over**: `chat.service.ts`, `chat-message-list.component.ts`, `chat.service.spec.ts`, `chat-page.component.spec.ts` are updated to use `msg.blocks` (interim inline text reduce). // Added to src/app/models/ai-chat.model.ts (DO NOT import @anthropic-ai/sdk here — model is SDK-agnostic per AI-SPEC.md §3 "Common Pitfall #2")
export interface TextBlock {
  type: 'text';
  text: string;
}

export interface ToolUseBlock {
  type: 'tool_use';
  id: string;
  name: string;
  input: unknown;
  status: 'pending' | 'approved' | 'discarded' | 'edited';
  editedFromText?: string;
}

export interface ToolResultBlock {
  type: 'tool_result';
  tool_use_id: string;
  content: string;
  isError?: boolean;
}

export type ChatBlock = TextBlock | ToolUseBlock | ToolResultBlock;

export interface AIToolSettings {
  enableDataQueryTools: boolean;
  enableMemoryTool: boolean;
  enableWebSearch: boolean;
  webSearchMaxUses: number;
  maxAgentTurns: number;

  redactWeightEntries: boolean;
  redactMealNotes: boolean;
}

};
